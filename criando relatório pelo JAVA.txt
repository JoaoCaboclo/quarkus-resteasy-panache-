@GetMapping("/relatorio/pedidos")
public ResponseEntity<String> gerarRelatorio(
        @RequestParam String clienteId,
        @RequestParam(required = false) String dataInicio,
        @RequestParam(required = false) String dataFim) {

    try {
        // 1. Buscar dados
        List<Pedido> pedidos = pedidoService.buscarPedidos(clienteId, dataInicio, dataFim);

        // 2. Caminhos do relatório
        String caminhoRelatorio = "relatorios/pedido.jasper";
        String caminhoSubRelatorio = "relatorios/itemPedido.jasper";

        // 3. Parametros para o Jasper
        Map<String, Object> parametros = new HashMap<>();
        parametros.put("CLIENTE_ID", clienteId);
        parametros.put("SUBREPORT_DIR", new File("relatorios/").getAbsolutePath() + File.separator);

        // 4. Fonte de dados do Jasper
        JRBeanCollectionDataSource dataSource = new JRBeanCollectionDataSource(pedidos);

        // 5. Preencher relatório
        JasperPrint jasperPrint = JasperFillManager.fillReport(
            new FileInputStream(caminhoRelatorio), parametros, dataSource);

        // 6. Gerar PDF
        String nomeArquivo = "relatorio_pedidos_" + UUID.randomUUID() + ".pdf";
        String caminhoPDF = "/caminho/onde/salvar/" + nomeArquivo;
        JasperExportManager.exportReportToPdfFile(jasperPrint, caminhoPDF);

        // 7. Retornar link (suponha que o serviço sirva arquivos de uma pasta)
        String urlArquivo = "http://seuservidor.com/arquivos/" + nomeArquivo;
        return ResponseEntity.ok(urlArquivo);

    } catch (Exception e) {
        e.printStackTrace();
        return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR)
                .body("Erro ao gerar relatório: " + e.getMessage());
    }
}

4. Relatório Controller
=======================

RelatorioController.java

@RestController
@RequestMapping("/relatorio")
public class RelatorioController {

    @Autowired
    private PedidoService pedidoService;

    @GetMapping("/pedidos")
    public ResponseEntity<String> gerarRelatorio(
        @RequestParam String clienteId,
        @RequestParam(required = false) String dataInicio,
        @RequestParam(required = false) String dataFim) {

        try {
            List<Pedido> pedidos = pedidoService.buscarPedidos(clienteId, dataInicio, dataFim);
            InputStream jasper = getClass().getResourceAsStream("/relatorios/pedido.jasper");

            Map<String, Object> parametros = new HashMap<>();
            parametros.put("SUBREPORT_DIR", new File("src/main/resources/relatorios/").getAbsolutePath() + File.separator);

            JRBeanCollectionDataSource dataSource = new JRBeanCollectionDataSource(pedidos);

            JasperPrint print = JasperFillManager.fillReport(jasper, parametros, dataSource);

            String nomeArquivo = "pedido_" + UUID.randomUUID() + ".pdf";
            String caminho = "arquivos/" + nomeArquivo;

            File out = new File(caminho);
            out.getParentFile().mkdirs();
            JasperExportManager.exportReportToPdfStream(print, new FileOutputStream(out));

            String url = "http://localhost:8080/arquivos/" + nomeArquivo;
            return ResponseEntity.ok(url);

        } catch (Exception e) {
            e.printStackTrace();
            return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR)
                .body("Erro: " + e.getMessage());
        }
    }

    @GetMapping("/arquivos/{arquivo}")
    public ResponseEntity<Resource> download(@PathVariable String arquivo) throws IOException {
        File file = new File("arquivos/" + arquivo);
        if (!file.exists()) return ResponseEntity.notFound().build();

        InputStreamResource resource = new InputStreamResource(new FileInputStream(file));
        return ResponseEntity.ok()
            .header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename=\"" + file.getName() + "\"")
            .contentType(MediaType.APPLICATION_PDF)
            .body(resource);
    }
}

package com.exemplo.relatorios.controller;

import com.exemplo.relatorios.model.Pedido;
import com.exemplo.relatorios.service.PedidoService;
import net.sf.jasperreports.engine.*;
import net.sf.jasperreports.engine.data.JRBeanCollectionDataSource;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.core.io.InputStreamResource;
import org.springframework.http.*;
import org.springframework.web.bind.annotation.*;

import java.io.*;
import java.util.*;

@RestController
@RequestMapping("/relatorio")
public class RelatorioController {

    @Autowired
    private PedidoService pedidoService;

    @GetMapping("/pdf-local")
    public ResponseEntity<String> gerarPdfLocal(@RequestParam String clienteId) {
        try {
            List<Pedido> pedidos = pedidoService.buscarPedidos(clienteId);
            JRBeanCollectionDataSource dataSource = new JRBeanCollectionDataSource(pedidos);

            JasperReport report = JasperCompileManager.compileReport(
                getClass().getResourceAsStream("/relatorios/pedido.jrxml")
            );

            Map<String, Object> params = new HashMap<>();
            params.put("SUBREPORT_DIR", getClass().getResource("/relatorios/").getPath());

            JasperPrint print = JasperFillManager.fillReport(report, params, dataSource);

            String nome = "pedido_" + UUID.randomUUID() + ".pdf";
            File pdf = new File("arquivos/" + nome);
            pdf.getParentFile().mkdirs();

            JasperExportManager.exportReportToPdfStream(print, new FileOutputStream(pdf));

            return ResponseEntity.ok("http://localhost:8080/arquivos/" + nome);
        } catch (Exception e) {
            return ResponseEntity.status(500).body("Erro: " + e.getMessage());
        }
    }

    @GetMapping("/pdf-base64")
    public ResponseEntity<String> gerarPdfBase64(@RequestParam String clienteId) {
        try {
            List<Pedido> pedidos = pedidoService.buscarPedidos(clienteId);
            JRBeanCollectionDataSource dataSource = new JRBeanCollectionDataSource(pedidos);

            JasperReport report = JasperCompileManager.compileReport(
                getClass().getResourceAsStream("/relatorios/pedido.jrxml")
            );

            Map<String, Object> params = new HashMap<>();
            params.put("SUBREPORT_DIR", getClass().getResource("/relatorios/").getPath());

            JasperPrint print = JasperFillManager.fillReport(report, params, dataSource);

            ByteArrayOutputStream baos = new ByteArrayOutputStream();
            JasperExportManager.exportReportToPdfStream(print, baos);

            String base64 = Base64.getEncoder().encodeToString(baos.toByteArray());
            return ResponseEntity.ok(base64);
        } catch (Exception e) {
            return ResponseEntity.status(500).body("Erro: " + e.getMessage());
        }
    }
}

@Service
public class RelatorioService {

    private final DataSource dataSource;

    public RelatorioService(DataSource dataSource) {
        this.dataSource = dataSource;
    }

    public byte[] gerarRelatorio(String templatePath, Map<String, Object> parametros) {
        try {
            // Carrega o template
            InputStream inputStream = new ClassPathResource(templatePath).getInputStream();
            JasperReport jasperReport = JasperCompileManager.compileReport(inputStream);

            // Prepara a fonte de dados
            Connection connection = dataSource.getConnection();

            // Prepara os parâmetros
            JRBeanCollectionDataSource dataSource = new JRBeanCollectionDataSource(parametros.get("listaDados"));

            // Gera o relatório
            JasperPrint jasperPrint = JasperFillManager.fillReport(jasperReport, parametros, connection);

            // Exporta para PDF
            byte[] pdfBytes = JasperExportManager.exportReportToPdf(jasperPrint);

            return pdfBytes;
        } catch (Exception e) {
            e.printStackTrace();
            return null;
        }
    }
}
